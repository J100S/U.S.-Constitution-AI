#!/usr/bin/env python3 """ Constitution Q&A - Flask app powered by Claude (Anthropic API) Run with: python app.py Then open: http://localhost:5000 """ import json import os from flask import Flask, request, jsonify, render_template_string import anthropic app = Flask(__name__) client = anthropic.Anthropic(api_key=os.environ.get("ANTHROPIC_API_KEY")) HTML = """ Constitution Q&A

Constitution Q&A

Powered by Artificial Intelligence

Welcome

Ask anything about the United States Constitution.

I'll find the relevant section and explain it.

First Amendment Congress Presidential Powers Rights if Arrested How Laws Are Made
""" SYSTEM_PROMPT = """You are a U.S. Constitution expert assistant. When given a question, respond ONLY with a JSON object (no markdown, no extra text) with these exact fields: { "section_name": "Name of the most relevant constitutional section (e.g., 'Article I, Section 8 — Powers of Congress' or 'First Amendment')", "section_text": "The actual constitutional text of that section (verbatim, as written in the Constitution)", "relevance_score": , "interpretation": "A clear 2-4 sentence plain-English explanation of what this section means and how it answers the question." } Always find the single most relevant section. Be precise with the constitutional text.""" @app.route("/") def index(): return render_template_string(HTML) @app.route("/ask", methods=["POST"]) def ask(): data = request.get_json() question = data.get("question", "").strip() if not question: return jsonify({"error": "No question provided."}), 400 try: message = client.messages.create( model="claude-sonnet-4-20250514", max_tokens=1000, system=SYSTEM_PROMPT, messages=[{"role": "user", "content": question}], ) raw = message.content[0].text.strip() # Strip markdown fences if present if raw.startswith("```"): raw = raw.split("```")[1] if raw.startswith("json"): raw = raw[4:] raw = raw.strip() result = json.loads(raw) return jsonify(result) except json.JSONDecodeError as e: return jsonify({"error": f"Failed to parse AI response: {e}"}), 500 except anthropic.AuthenticationError: return jsonify({"error": "Invalid API key. Set ANTHROPIC_API_KEY environment variable."}), 401 except Exception as e: return jsonify({"error": str(e)}), 500 if __name__ == "__main__": print("=" * 55) print(" 🇺🇸 Constitution Q&A — http://localhost:5000") print("=" * 55) if not os.environ.get("ANTHROPIC_API_KEY"): print("\n ⚠️ WARNING: ANTHROPIC_API_KEY is not set!") print(" Export it before running:\n") print(" export ANTHROPIC_API_KEY=sk-ant-...") print() app.run(debug=True, port=5000)
